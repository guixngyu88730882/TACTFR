        private static string ReadIniValue(string section, string key, string fallback)
        {
            try
            {
                if (!File.Exists(ConfigPath)) return fallback;
                string currentSection = "";
                foreach (var rawLine in File.ReadAllLines(ConfigPath))
                {
                    string line = rawLine.Trim();
                    if (string.IsNullOrEmpty(line)) continue;
                    if (line.StartsWith(";") || line.StartsWith("#")) continue;

                    if (line.StartsWith("[") && line.EndsWith("]"))
                    {
                        currentSection = line.Substring(1, line.Length - 2).Trim();
                        continue;
                    }

                    if (!string.Equals(currentSection, section, StringComparison.OrdinalIgnoreCase))
                        continue;

                    int eq = line.IndexOf('=');
                    if (eq < 0) continue;

                    string k = line.Substring(0, eq).Trim();
                    string v = line.Substring(eq + 1).Trim();
                    int commentIdx = v.IndexOf(';');
                    if (commentIdx >= 0) v = v.Substring(0, commentIdx).Trim();

                    if (string.Equals(k, key, StringComparison.OrdinalIgnoreCase))
                        return v;
                }
            }
            catch { }
            return fallback;
        }

        private static Keys ParseKey(string value, Keys fallback)
        {
            if (string.IsNullOrWhiteSpace(value)) return fallback;
            try
            {
                if (Enum.TryParse(value.Trim(), true, out Keys parsed))
                    return parsed;
            }
            catch { }
            return fallback;
        }

        private static float ParseFloat(string value, float fallback)
        {
            if (string.IsNullOrWhiteSpace(value)) return fallback;
            if (float.TryParse(value.Trim(), NumberStyles.Float, CultureInfo.InvariantCulture, out float parsed))
                return parsed;
            if (float.TryParse(value.Trim(), NumberStyles.Float, CultureInfo.CurrentCulture, out parsed))
                return parsed;
            return fallback;
        }

        private static void LoadTerminalConfig()
        {
            TerminalInteractRange = Math.Max(1.5f, ParseFloat(ReadIniValue("PoliceTerminal", "InteractRange", "3.2"), 3.2f));
            string rawPositions = ReadIniValue("PoliceTerminal", "Positions", "441.4,-981.8,30.7|443.5,-981.0,30.7");
            var parsedPositions = ParseVectorList(rawPositions);
            if (parsedPositions.Count > 0)
            {
                _terminalPositions.Clear();
                _terminalPositions.AddRange(parsedPositions);
            }
        }

        private static List<Vector3> ParseVectorList(string raw)
        {
            var result = new List<Vector3>();
            if (string.IsNullOrWhiteSpace(raw)) return result;

            string[] items = raw.Split(new[] { '|' }, StringSplitOptions.RemoveEmptyEntries);
            for (int i = 0; i < items.Length; i++)
            {
                string[] parts = items[i].Split(new[] { ',' }, StringSplitOptions.RemoveEmptyEntries);
                if (parts.Length != 3) continue;

                float x = ParseFloat(parts[0], float.NaN);
                float y = ParseFloat(parts[1], float.NaN);
                float z = ParseFloat(parts[2], float.NaN);
                if (float.IsNaN(x) || float.IsNaN(y) || float.IsNaN(z)) continue;

                result.Add(new Vector3(x, y, z));
            }

            return result;
        }

        private static void ValidateKeyBindings()
        {
            bool hasConflict = false;

            // Check OpenTerminal conflicts
            if (KeyBindings.OpenTerminal == KeyBindings.MenuCancel)
            {
                ModLog.Warn("[Config] OpenTerminal conflicts with MenuCancel; resetting MenuCancel to Backspace");
                KeyBindings.MenuCancel = Keys.Back;
                hasConflict = true;
            }

            if (KeyBindings.OpenTerminal == KeyBindings.MenuConfirm)
            {
                ModLog.Warn("[Config] OpenTerminal conflicts with MenuConfirm; resetting MenuConfirm to NumPad5");
                KeyBindings.MenuConfirm = Keys.NumPad5;
                hasConflict = true;
            }

            // Check VehicleTerminal conflicts
            if (KeyBindings.VehicleTerminal == KeyBindings.MenuCancel)
            {
                ModLog.Warn("[Config] VehicleTerminal conflicts with MenuCancel; resetting MenuCancel to Backspace");
                KeyBindings.MenuCancel = Keys.Back;
                hasConflict = true;
            }

            // Check PatrolMenu conflicts
            if (KeyBindings.PatrolMenu == KeyBindings.MenuCancel)
            {
                ModLog.Warn("[Config] PatrolMenu conflicts with MenuCancel; resetting MenuCancel to Backspace");
                KeyBindings.MenuCancel = Keys.Back;
                hasConflict = true;
            }

            // Check DispatchMenu conflicts
            if (KeyBindings.DispatchMenu == KeyBindings.MenuCancel)
            {
                ModLog.Warn("[Config] DispatchMenu conflicts with MenuCancel; resetting MenuCancel to Backspace");
                KeyBindings.MenuCancel = Keys.Back;
                hasConflict = true;
            }

            // Check OpenTerminal vs VehicleTerminal conflict
            if (KeyBindings.OpenTerminal == KeyBindings.VehicleTerminal)
            {
                ModLog.Warn("[Config] OpenTerminal conflicts with VehicleTerminal; resetting VehicleTerminal to T");
                KeyBindings.VehicleTerminal = Keys.T;
                hasConflict = true;
            }

            if (hasConflict)
            {
                ModLog.Info("[Config] Key binding conflicts were detected and automatically resolved");
            }
        }

        private static void LoadLogging()

            if (int.TryParse(maxSize.Trim(), out int kb) && kb >= 0)
                LogMaxSizeKB = kb;
        }

using System;
using System.Collections.Generic;

namespace EF.PoliceMod.Core
{
    public static class EventBus
    {

          
              private static readonly Dictionary<Type, List<Delegate>> _listeners =
     new Dictionary<Type, List<Delegate>>();
        private static readonly object _listenersLock = new object();

        public static void Unsubscribe<T>(Action<T> handler)
        {
            var type = typeof(T);
            lock (_listenersLock)
            {
                if (!_listeners.ContainsKey(type)) return;

                _listeners[type].Remove(handler);

                if (_listeners[type].Count == 0)
                    _listeners.Remove(type);
            }
        }


        public static void ClearAll()
        {
            lock (_listenersLock)
            {
                _listeners.Clear();
            }
        }
        public static void Subscribe<T>(Action<T> handler)
        {
            var type = typeof(T);
            lock (_listenersLock)
            {
                if (!_listeners.TryGetValue(type, out var list))
                {
                    list = new List<Delegate>();
                    _listeners[type] = list;
                }

                if (!list.Contains(handler))
                {
                    list.Add(handler);
                }
            }
        }

        public static void Publish<T>(T message)
        {
            var type = typeof(T);

#if DEBUG
            ModLog.Info($"Event Published: {type.Name}");
#endif

            List<Delegate> handlersSnapshot;
            lock (_listenersLock)
            {
                if (!_listeners.TryGetValue(type, out var list) || list == null || list.Count == 0)
                {
                    ModLog.Warn($"No listeners for event: {type.Name}");
                    return;
                }
                handlersSnapshot = new List<Delegate>(list);
            }

            foreach (var handler in handlersSnapshot)
            {
                try
                {
                    if (handler is Action<T> actionT)
                    {
                        actionT(message);
                    }
                    else
                    {
                        handler.DynamicInvoke(message);
                    }
                }
                catch (Exception ex)
                {
                    ModLog.Error($"Event handler crash: {type.Name} | {ex}");
                }
            }
        }
    }
}

    public class EFCore : Script

        // ... 字段定义省略 ...

        private void OnTick(object sender, EventArgs e)
        {
            Ped player = Game.Player.Character;

            // 玩家死亡时由 DutyLifecycleController / PlayerRespawnFixSystem 处理，
            // 这里不要每帧广播 DutyEndedEvent，避免事件风暴。

            if (_lawAuthorityActive)
            {
                int now = Game.GameTime;
                if (now - _lastWantedEnforceAtMs >= WantedEnforceIntervalMs)
                {
                    _lastWantedEnforceAtMs = now;
                    try { EnableLawAuthority(); } catch { }
                }

                try
                {
                    if (Game.Player.WantedLevel > 0)
                        Game.Player.WantedLevel = 0;
                }
                catch { }
            }

            // === 执行顺序 ===
            // 1. InputManager.Update() - 处理按键输入
            try { _input?.Update(); } catch (Exception ex) { ModLog.Error("[EFCore] Input.Update exception: " + ex); }

            // 2. Duty controller
            try { _dutyController?.Tick(); } catch (Exception ex) { ModLog.Error("[EFCore] DutyController.Tick exception: " + ex); }

            // 3. LockTargetSystem
            try { LockTargetSystem?.Update(); } catch (Exception ex) { ModLog.Error("[EFCore] LockTargetSystem.Update exception: " + ex); }

            try { _deliverSystem?.Update(); } catch (Exception ex) { ModLog.Error("[EFCore] DeliverSystem.Update exception: " + ex); }
            try { _pullOverSystem?.TickUpdate(); } catch (Exception ex) { ModLog.Error("[EFCore] PullOverSystem.TickUpdate exception: " + ex); }

            // 6. Menus (DispatchMenu, HeliRecon, ArrestMenu, etc.)
            try { _dispatchMenu?.Tick(); } catch (Exception ex) { ModLog.Error("[EFCore] DispatchMenu.Tick exception: " + ex); }
            try { _heliRecon?.Tick(); } catch (Exception ex) { ModLog.Error("[EFCore] HeliRecon.Tick exception: " + ex); }
